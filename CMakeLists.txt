cmake_minimum_required(VERSION 3.20)
project(babel_meaning_miner VERSION 0.3.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BMM_BUILD_TESTS "Build tests" ON)

find_package(Threads REQUIRED)
find_package(SQLite3 QUIET)
if(NOT SQLite3_FOUND)
  add_library(sqlite3 INTERFACE)
  target_include_directories(sqlite3 INTERFACE third_party/sqlite-unpack/sqlite-amalgamation-3540100)
  add_library(SQLite::SQLite3 ALIAS sqlite3)
  set(BMM_SQLITE_SOURCE third_party/sqlite-unpack/sqlite-amalgamation-3540100/sqlite3.c)
endif()
add_library(bmm_core OBJECT
  src/generator/generator.cpp src/analyzer/analyzer.cpp
  src/analyzer/sentence.cpp src/analyzer/language_model.cpp
  src/analyzer/passage.cpp src/experiment/feasibility.cpp src/storage/passages.cpp
  src/storage/storage.cpp src/scanner/scanner.cpp ${BMM_SQLITE_SOURCE})
target_include_directories(bmm_core PUBLIC include)
target_link_libraries(bmm_core PUBLIC SQLite::SQLite3 Threads::Threads)
if(MSVC)
  target_compile_options(bmm_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(bmm_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(bmm src/main.cpp src/cli/v03.cpp)
target_link_libraries(bmm PRIVATE bmm_core)
install(TARGETS bmm RUNTIME DESTINATION bin)

include(CTest)
if(BMM_BUILD_TESTS)
  add_executable(bmm_tests tests/tests.cpp)
  target_link_libraries(bmm_tests PRIVATE bmm_core)
  add_test(NAME bmm_tests COMMAND bmm_tests)
  set_tests_properties(bmm_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  foreach(suite passage_v03 model_v03 recovery_v03)
    add_executable(${suite} tests/${suite}.cpp)
    target_link_libraries(${suite} PRIVATE bmm_core)
    add_test(NAME ${suite} COMMAND ${suite})
    set_tests_properties(${suite} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 60)
  endforeach()
  find_program(BMM_PY_LAUNCHER py)
  if(BMM_PY_LAUNCHER)
    set(BMM_PYTHON_COMMAND ${BMM_PY_LAUNCHER} -3)
  else()
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
      set(BMM_PYTHON_COMMAND ${Python3_EXECUTABLE})
    endif()
  endif()
  if(BMM_PYTHON_COMMAND)
    add_test(NAME semantic_adapter COMMAND ${BMM_PYTHON_COMMAND} ${CMAKE_SOURCE_DIR}/tests/test_semantic.py)
    add_test(NAME cli_recovery COMMAND ${BMM_PYTHON_COMMAND} ${CMAKE_SOURCE_DIR}/tests/test_cli_v03.py $<TARGET_FILE:bmm>)
    set_tests_properties(cli_recovery PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 60)
  endif()
endif()
